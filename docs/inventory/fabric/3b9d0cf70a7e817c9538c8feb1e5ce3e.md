## Local Deployment (Docker Desktop)
Run locally to review features and concepts. Clone from GitHub link above, or use the pre-built onemindos image.
### docker-compose.yml
```yaml
version: '3.8'
services:
  worldmonitor:
    image: ghcr.io/onemindos/worldmonitor:latest
    # OR build from source:
    # build: .  # clone https://github.com/koala73/worldmonitor
    ports:
      - "8080:8080"
    environment:
      UPSTASH_REDIS_REST_URL: "http://worldmonitor-redis-rest:80"
      UPSTASH_REDIS_REST_TOKEN: "your-redis-token"  # set in .env
      LOCAL_API_PORT: "46123"
      LOCAL_API_MODE: "docker"
      LOCAL_API_CLOUD_FALLBACK: "false"
      WS_RELAY_URL: "http://worldmonitor-ais-relay:3004"
      AISSTREAM_API_KEY: "your-aisstream-key"  # https://aisstream.io
    depends_on:
      - worldmonitor-redis
      - worldmonitor-redis-rest
      - worldmonitor-ais-relay
    restart: unless-stopped

  worldmonitor-ais-relay:
    image: ghcr.io/onemindos/worldmonitor-relay:latest
    ports:
      - "3004:3004"
    restart: unless-stopped

  worldmonitor-redis:
    image: redis:7-alpine
    volumes:
      - worldmonitor-redis-data:/data
    restart: unless-stopped

  worldmonitor-redis-rest:
    image: ghcr.io/onemindos/worldmonitor-redis-rest:latest
    ports:
      - "8081:80"
    depends_on:
      - worldmonitor-redis
    restart: unless-stopped

volumes:
  worldmonitor-redis-data:
```
### Quick Start
```shell
# Save the above as docker-compose.yml then:
docker compose up -d

# View logs
docker compose logs -f

# Stop
docker compose down
```
