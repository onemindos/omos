### The Scheduling Algorithm

```python
def schedule_task(task: Item, user: User) -> datetime:
    """
    OneMind Intelligent Scheduling

    Unlike traditional apps that just track due dates,
    OneMind calculates the OPTIMAL time to work on each task.
    """

    # 1. Gather context
    calendar = get_calendar(user)
    energy_patterns = get_energy_patterns(user)  # From biometrics
    current_workload = get_workload(user)
    path_balance = get_path_balance(user)  # HP/LE/GE allocation

    # 2. Determine task requirements
    duration = task.metadata.get('duration', 30)  # minutes
    energy_required = task.metadata.get('energy_required', 'medium')
    deadline = task.metadata.get('due_date')
    path = task.path

    # 3. Find optimal slot
    candidates = []

    for slot in calendar.get_free_slots():
        score = 0

        # Energy match (high energy tasks in peak hours)
        if energy_matches(slot.time, energy_required, energy_patterns):
            score += 30

        # Path balance (favor neglected paths)
        if path_needs_attention(path, path_balance):
            score += 20

        # Buffer before deadline
        days_before_deadline = (deadline - slot.date).days
        if days_before_deadline >= 2:
            score += 25
        elif days_before_deadline == 1:
            score += 10
        else:
            score -= 50  # Panic zone

        # Workload distribution
        if slot.date not in overloaded_days(current_workload):
            score += 15

        candidates.append((slot, score))

    # 4. Return best slot
    best_slot = max(candidates, key=lambda x: x[1])
    return best_slot[0].datetime
```

## Part 10: Database Schema

### Universal Item Table

```sql
-- ═══════════════════════════════════════════════════════════════════════════
--                         UNIVERSAL ITEM TABLE
-- ═══════════════════════════════════════════════════════════════════════════

CREATE TYPE item_type AS ENUM (
    'task', 'habit', 'project', 'doc', 'recipe', 'goal',
    'skill', 'workflow', 'note', 'contact', 'event', 'metric'
);

CREATE TYPE path_type AS ENUM ('HP', 'LE', 'GE');

CREATE TYPE item_status AS ENUM ('inbox', 'active', 'done', 'archived', 'deleted');

CREATE TYPE workforce_type AS ENUM ('self', 'legacy', 'physical', 'hive');

CREATE TABLE items (
    -- Core fields (all items)
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         VARCHAR(50) NOT NULL,
    type            item_type NOT NULL,
    title           VARCHAR(500) NOT NULL,
    content         TEXT,  -- Markdown body
    path            path_type NOT NULL,
    status          item_status DEFAULT 'inbox',
    assignee        workforce_type DEFAULT 'self',

    -- Hierarchy
    parent_id       UUID REFERENCES items(id) ON DELETE SET NULL,

    -- Type-specific metadata (JSONB for flexibility)
    metadata        JSONB DEFAULT '{}',

    -- Timestamps
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW(),
    completed_at    TIMESTAMPTZ,
    deleted_at      TIMESTAMPTZ,

    -- Scheduling (Motion-style)
    due_date        DATE,
    do_date         TIMESTAMPTZ,  -- AI-calculated optimal time
    duration        INT,  -- minutes
    priority        VARCHAR(10),
    energy_required VARCHAR(10),

    -- Search
    search_vector   TSVECTOR GENERATED ALWAYS AS (
        setweight(to_tsvector('english', coalesce(title, '')), 'A') ||
        setweight(to_tsvector('english', coalesce(content, '')), 'B')
    ) STORED
);

-- Indexes
CREATE INDEX idx_items_user_path ON items(user_id, path);
CREATE INDEX idx_items_user_type ON items(user_id, type);
CREATE INDEX idx_items_user_status ON items(user_id, status);
CREATE INDEX idx_items_parent ON items(parent_id);
CREATE INDEX idx_items_due_date ON items(due_date) WHERE due_date IS NOT NULL;
CREATE INDEX idx_items_do_date ON items(do_date) WHERE do_date IS NOT NULL;
CREATE INDEX idx_items_search ON items USING GIN(search_vector);

-- ═══════════════════════════════════════════════════════════════════════════
--                         ITEM METADATA EXAMPLES
-- ═══════════════════════════════════════════════════════════════════════════

-- Task metadata:
-- {
--   "priority": "high",

--   "energy_required": "high",
--   "recurring": false,
--   "tags": ["workout", "morning"]
-- }

-- Habit metadata:
-- {
--   "frequency": "daily",
--   "time_of_day": "06:00",
--   "streak": 23,
--   "cue": "Alarm goes off",
--   "reward": "Coffee"
-- }

-- Recipe metadata:
-- {
--   "ingredients": ["chicken", "carrots", "celery"],
--   "steps": ["Boil chicken", "Add vegetables"],
--   "servings": 8,
--   "prep_time": 20,

-- Skill metadata:
-- {

--   "practice_hours": 120,
--   "resources": ["Book: XYZ", "Course: ABC"]
-- }

-- ═══════════════════════════════════════════════════════════════════════════
--                         INTELLIGENCE SCORES
-- ═══════════════════════════════════════════════════════════════════════════

CREATE TABLE intelligence_scores (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         VARCHAR(50) NOT NULL,
    score_type      VARCHAR(10) NOT NULL CHECK (score_type IN ('HPI', 'LEI', 'GEI', 'UI')),
    score           DECIMAL(5,2) NOT NULL CHECK (score >= 0 AND score <= 100),
    components      JSONB NOT NULL DEFAULT '{}',
    period_start    DATE NOT NULL,
    period_end      DATE NOT NULL,
    calculated_at   TIMESTAMPTZ DEFAULT NOW(),
    created_at      TIMESTAMPTZ DEFAULT NOW(),

    UNIQUE(user_id, score_type, period_start)
);

CREATE INDEX idx_scores_user_type ON intelligence_scores(user_id, score_type);
CREATE INDEX idx_scores_period ON intelligence_scores(period_start, period_end);

-- ═══════════════════════════════════════════════════════════════════════════
--                         FAMILY MEMBERS
-- ═══════════════════════════════════════════════════════════════════════════

CREATE TABLE family_members (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         VARCHAR(50) NOT NULL,  -- Parent/owner user
    member_id       VARCHAR(50) UNIQUE,     -- Their own user_id if they have one
    name            VARCHAR(255) NOT NULL,
    relationship    VARCHAR(50) NOT NULL,   -- child, spouse, parent
    birth_date      DATE,
    has_own_account BOOLEAN DEFAULT false,
    avatar_url      TEXT,
    created_at      TIMESTAMPTZ DEFAULT NOW()
);
```
